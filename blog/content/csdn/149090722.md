一个经典的套路，二分答案+check,检查是否可行也是经典的转化，把符合条件的变成1，不符合条件的置0，现在问题就变成了一个01序列可以用中位数代替3个数,让0尽量消去，1尽量多。

首先可以贪心的想到的是三个0变成一个0肯定是比别的好的。

对于连续存在的超过两个的0，任选其中3个0变成1个0，剩下一定是连续的两个0或者一个0.

特别的00100这种情况，最终可以变成一个0，先把001 $\rightarrow$ 0,再把 $000\rightarrow 0$

对于剩下的一般情况，我只要考虑最后是不是1的数量大于0的数量即可,因为最后的0只能单独出现，我们可以找到 101,011,110这种组合用1个1消掉一个0

频繁使用vector的emplace_back 和 pop_back 时间会慢，可以考虑手写stack



```cpp
#pragma GCC optimize(3,"Ofast","inline")
#include<bits/stdc++.h>
using namespace std;
using i64 = long long;
using i128 = __int128;

int n;
void solve(){
    cin>>n;
    vector<int> a(n);
    for(auto &i:a) cin>>i;
    auto check =[&](int x)->bool{
        vector<int> st;
        array<int,3> tmp;
        for(int i = 0;i<n;++i){
            st.emplace_back(a[i]>=x);
            if(st.size()>=3){
                int cnt = 0;
                for(int j=0;j<3;++j){
                    cnt+=st.back();
                    tmp[2-j] = st.back();
                    st.pop_back();
                }
                if(cnt>=1){
                    if(tmp==array<int,3>{0,0,1}) st.emplace_back(0);
                    else for(auto &j:tmp) st.emplace_back(j);
                }else st.emplace_back(0);
            }
        }
        int cnt=0;
        for(auto &i:st) cnt+=i;
        return cnt>st.size()-cnt;
    };
    int l = 1,r = 1e9;
    while(l<=r){
        int mid =(l+r)>>1;
        if(check(mid)) l = mid+1;
        else r = mid-1;
    }
    cout<<r<<"\n";
}

signed main(){
    ios::sync_with_stdio(0);cin.tie(0);
    int t=1;
    cin>>t;
    while(t--) solve();
    return 0;
}
```
