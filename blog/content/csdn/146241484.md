首先不考虑删除操作,那么插入元素这件事情就是DP(背包或者可达DP) 做的，设 $DP_i$ 为能否用S中的元素组合得到 $i$ 具体的转移式子为 $dp_i = dp_i|dp_{i-x}$,这样的时间复杂度是 $O(ns)$

考虑**优化**，每次转移只与上一次的状态有关且转移的偏移量相同，那么可以用 $bitset$ 对状态转移进行优化, $dp_i|=dp_i<<x$ ,时间复杂度 $O(\frac{ns}{w})$

引入删除操作，由于单纯的使用 $bitset$ 不方便维护,这里我学到了一种技巧，**将删除操作看成DFS的回退操作** ，引入一棵时间的线段树进行维护每个时间对应的区间的信息,即将加入和删除操作看成一段时间的加入，然后进行DFS,就可以 $log$ 次求出所有答案


```cpp
#pragma GCC optimize(3,"Ofast","inline")
#include<bits/stdc++.h>
using namespace std;
using i64 = long long;
using i128 = __int128;

constexpr int maxn = 5e5+10;
int n,op,x,ans[maxn];
vector<int> p[maxn],date[maxn<<2];
bitset<maxn> now;

void update(int p,int l,int r,int x,int y,int num){
    if(x>r||y<l) return;
    if(x<=l&&r<=y){
        date[p].emplace_back(num);
        return;
    }
    int mid = (l+r)>>1;
    update(p*2,l,mid,x,y,num);
    update(p*2+1,mid+1,r,x,y,num);
}

void dfs(int p,int l,int r){
    bitset<maxn> tmp = now;
    for(auto &i:date[p]) now|=(now<<i);
    if(l==r) ans[l] = now.count();
    else{
        int mid = (l+r)>>1;
        dfs(p*2,l,mid);
        dfs(p*2+1,mid+1,r);
    }
    now = tmp;
}

signed main(){
    ios::sync_with_stdio(0);cin.tie(0);
    cin>>n;
    for(int i = 1;i<=n;++i){
        cin>>op>>x;
        if(op==1){ 
            p[x].emplace_back(i);
        }else{
            update(1,1,n,p[x].back(),i-1,x);
            p[x].pop_back();
        }
    }
    for(int i= 1;i<=500000;++i){
        if(p[i].empty()) continue;
        while(!p[i].empty()){
            update(1,1,n,p[i].back(),n,i);
            p[i].pop_back();
        }
    }
    now.set(0);//将第零位设置成1
    dfs(1,1,n);
    for(int i = 1;i<=n;++i){
        cout<<ans[i]-1<<'\n';//因为把一位设置成1了所以需要减掉1
    }
    return 0;
}     
```
