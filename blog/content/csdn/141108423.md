总的来说就是对$\leq lim$的$a_i$让它用第一种方式去更新,因为小的数更容易被更新,对于$>lim$的$b_i$用他们去更新$a_i$(第二种方式),因为大的数更容易更新别的数

**several supplement:**

**1.两种维护手法的时间复杂度为什么是O(x)?**
大概是因为两种维护方法都与b的第$x$大元素有关

**2.为什么操作期望次数是那样子的?**
首先我们需要知道，$\frac{1}{1-x}=\sum_{n=0}^{\infty} x^{n}$,而$1-\frac{x}{n}$取个倒数才是我们需要的形式,指数是i+1，所以相当于从1开始，所以$\approx\frac{n}{x}$

```cpp
#include<bits/stdc++.h>
using namespace std;
using i64 = long long;
using i128 = __int128;
#define ios ios::sync_with_stdio(0);cin.tie(0);cout.tie(0);

//取模运算可以看做4次加减法运算
int n,q,lim,ca,cb;
inline int add(int x,int k){
    if(x+k>=n) return x+k-n;
    return x+k;
}
inline int minu(int x,int k){
    if(x-k<0) return x-k+n;
    return x-k;
}

inline void solve(){
    i64 ans = 0;
    cin>>n>>q;
    vector<int>a(n),b(n),op(n);
    for(auto &i:a) cin>>i,ans+=i;
    for(int i = 0;i<n;++i){
        cin>>b[i];
        op[i] = b[i];
    }
    lim = sqrt(n);
    lim = max(1,lim);
    lim = min(n,lim);
    sort(op.begin(),op.end(),greater<int>());
    lim = op[lim-1];ca=0,cb=0;
    vector<int> oa(n),ob(n);
    for(int i = 0;i<n;++i){
        if(a[i]<=lim) oa[ca++] = i;
        if(b[i]>lim)  ob[cb++] = i;
    }
    while(q--){
        int k;cin>>k;
        for(int i = 0;i<ca;){//因为用了swap,i不可以写在这里,会改变当前i指向的位置而导致少更新
            int t = oa[i];
            if(b[add(t,k)]>a[t]){
                ans+=b[add(t,k)]-a[t];
                a[t] = b[add(t,k)];
            }
            if(a[t]>lim) swap(oa[--ca],oa[i]);
            else ++i;
        }
        for(int i = 0;i<cb;++i){
            int t = ob[i];
            if(a[minu(t,k)]<b[t]){
                ans+=(b[t]-a[minu(t,k)]);
                a[minu(t,k)] = b[t];
            }
        }
        cout<<ans<<"\n";
    }
}


signed main(){
    ios;
    int t;cin>>t;
    while(t--){
        solve();
    }
    return 0;
}
```
